-- Table: public.address

-- DROP TABLE IF EXISTS public.address;

CREATE TABLE IF NOT EXISTS public.address
(
    id integer NOT NULL DEFAULT nextval('address_id_seq'::regclass),
    region character varying(100) COLLATE pg_catalog."default" NOT NULL,
    city character varying(100) COLLATE pg_catalog."default" NOT NULL,
    street character varying(100) COLLATE pg_catalog."default" NOT NULL,
    building_number character varying(10) COLLATE pg_catalog."default" NOT NULL,
    apartment_number integer,
    CONSTRAINT address_pkey PRIMARY KEY (id),
    CONSTRAINT region_format CHECK (region::text ~ '^[А-ЯІЇЄа-яіїє]+$'::text),
    CONSTRAINT city_format CHECK (city::text ~ '^[А-ЯІЇЄа-яіїє\-]+$'::text),
    CONSTRAINT street_format CHECK (street::text ~ '^[А-ЯІЇЄа-яіїє\-]+$'::text),
    CONSTRAINT building_number_format CHECK (building_number::text ~ '^[0-9\/]+$'::text),
    CONSTRAINT apartment_number_format CHECK (apartment_number > 0)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.address
    OWNER to postgres;

-- Table: public.case

-- DROP TABLE IF EXISTS public."case";

CREATE TABLE IF NOT EXISTS public."case"
(
    id integer NOT NULL DEFAULT nextval('case_id_seq'::regclass),
    client_id integer NOT NULL,
    detective_id integer NOT NULL,
    case_type_id integer NOT NULL,
    title character varying(255) COLLATE pg_catalog."default" NOT NULL,
    description text COLLATE pg_catalog."default" NOT NULL,
    start_date date NOT NULL DEFAULT CURRENT_DATE,
    close_date date,
    deadline_date date NOT NULL,
    status case_status,
    CONSTRAINT case_pkey PRIMARY KEY (id),
    CONSTRAINT case_case_type_id_fkey FOREIGN KEY (case_type_id)
        REFERENCES public.case_type (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT case_client_id_fkey FOREIGN KEY (client_id)
        REFERENCES public.client (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT case_detective_id_fkey FOREIGN KEY (detective_id)
        REFERENCES public.detective (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT close_date_format CHECK (close_date >= start_date),
    CONSTRAINT deadline_format CHECK (deadline_date >= start_date) NOT VALID
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public."case"
    OWNER to postgres;
